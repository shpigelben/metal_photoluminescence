\section{Electronic Density of States}
\label{app:electronic_density_of_states}

The change in variables in 3D goes as follows
\begin{equation}
d^{3}k\to 4\pi k^{2} dk \to \underbrace{ 4\pi k^{2} \frac{dk}{d\mathcal{E}} }_{ \rho(\mathcal{E}) } d\mathcal{E}
\label{eq:edos_change_of_vars}
\end{equation}
the momentum energy relation comes from the dispersion relation of the electron which is approximated to be that of a free electron for a conduction electron in the 1st BZ.

\begin{equation}
\mathcal{E}(k)=\frac{\hbar^{2}k^{2}}{2m}\quad \to \quad k(\mathcal{E}) = \frac{\sqrt{ 2m\mathcal{E} }}{\hbar} \quad \to \quad \frac{dk}{d\mathcal{E}}=\frac{1}{\hbar}\sqrt{ \frac{m}{2\mathcal{E}} } 
\label{eq:edos_dispersion_relation}
\end{equation}
plugging the relevant expressions in Eq.~\eqref{eq:edos_dispersion_relation} into Eq.~\eqref{eq:edos_change_of_vars} results in

\begin{equation}
\rho(\mathcal{E}) = \frac{4\pi {m_e}^{3/2}}{\hbar^3}\sqrt{2\mathcal{E}}
\label{eq:edos_free_electron}
\end{equation}

\subsection{A general Form of DOS}
\begin{align}
\rho(\mathcal{E}) &= \int\limits_{\text{BZ}} \, d^{3}k \; \delta\Big(\mathcal{E}-\mathcal{E}(\mathbf{k})\Big)  \\
&= \int\limits_{\Omega}  \, d^{3}k' \; \frac{\delta(\mathbf{k}-\mathbf{k}')}{|\nabla_{\mathbf{k}} \mathcal{E}|}
\label{eq:edos_general_form}
\end{align}
Where $\Omega$ is the surface on which $\mathcal{E}(\mathbf{k})=\mathcal{E}$
