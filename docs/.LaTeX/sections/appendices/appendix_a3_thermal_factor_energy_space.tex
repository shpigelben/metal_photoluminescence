\section{Thermal Factor Energy Space}
\label{app:thermal_factor_energy_space}

\newpage
\subsection{Equilibrium}
\begin{align}
I & = \; \int\limits_{0}^{\infty} f(\mathcal{E}+\hbar\omega)[1-f(\mathcal{E})]d\mathcal{E}
\label{eq:thermal_factor_1}\\
&= \; n_{B}(\hbar \omega) \left[ \int\limits_{0}^{\infty}f(\mathcal{E})d\mathcal{E}-\int\limits_{0}^{\infty}f(\mathcal{E}+\hbar\omega) d\mathcal{E}\right]
\label{eq:thermal_factor_2}\\
&= \; n_{B}(\hbar \omega) \left[ \int\limits_{0}^{\infty}f(\mathcal{E})d\mathcal{E}-\int\limits_{\hbar\omega}^{\infty}f(\mathcal{E}) d\mathcal{E}\right]
\label{eq:thermal_factor_3}\\
&= \; n_{B}(\hbar \omega) \int\limits_{0}^{\hbar\omega}f(\mathcal{E})d\mathcal{E}
\label{eq:thermal_factor_4}\\
&= \; n_{B}(\hbar \omega) \int\limits_{0}^{\hbar\omega} \frac{1}{e^{\beta(\mathcal{E}-\mu)}+1}d\mathcal{E}
\label{eq:thermal_factor_5}\\
&= \; n_{B}(\hbar \omega) \int\limits_{0}^{\hbar\omega} \left[ 1+ \frac{e^{\beta(\mathcal{E}-\mu)}}{e^{\beta(\mathcal{E}-\mu)}+1} \right] d\mathcal{E}
\label{eq:thermal_factor_6}\\
&= \; n_{B}(\hbar \omega) \left[ \mathcal{E}- \frac{1}{\beta}\ln \left(e^{\beta(\mathcal{E}-\mu)}+1\right) \right]_{\mathcal{E=0}}^{\mathcal{E}=\hbar\omega}
\label{eq:thermal_factor_7}\\
&= \; n_{B}(\hbar \omega)  \left[ \hbar\omega+ \frac{1}{\beta}\ln\left(1+e^{-\beta\mu}\right)-\frac{1}{\beta}\ln\left(1+e^{\beta(\hbar\omega-\mu)}\right) \right]
\label{eq:thermal_factor_8}\\
& \approx \; \frac{\hbar\omega}{e^{\beta \hbar\omega}-1}
\label{eq:thermal_factor_9}
\end{align}

The transition from Eq.~\eqref{eq:thermal_factor_1} to Eq.~\eqref{eq:thermal_factor_2} is a known identity
\begin{equation}
f(\mathcal{E}+\hbar\omega) [1-f(\mathcal{E})] = n_{B}(\hbar\omega) \left[f(\mathcal{E})-f(\mathcal{E+\hbar \omega})\right]
\label{eq:thermal_factor_identity}
\end{equation}
where $n_{B}(\hbar\omega)$ is the Bose-Einstein distribution

\begin{equation}
n_{B}(\hbar\omega) = \frac{1}{e^{\beta \hbar\omega}-1}
\label{eq:thermal_factor_bose_einstein}
\end{equation}
The approximation from Eq.~\eqref{eq:thermal_factor_8} to Eq.~\eqref{eq:thermal_factor_9} is valid when the following conditions are met
\begin{equation}
\mu \gg k_{\small B}T \quad\text{and}\quad \mu\gg \hbar\omega
\label{eq:thermal_factor_conditions}
\end{equation}

\subsection{Non-equilibrium}

\begin{equation}
f\big( { \mathcal{E} \ | \ \omega _{\scriptsize L} }\big) = f^{T}({ \mathcal{E}}) + \delta E(\omega _{\scriptsize L})\cdot B({ \mathcal{E} \ | \ \omega _{\scriptsize L}}) 
\label{eq:thermal_factor_noneq_distribution}
\end{equation}
\begin{equation}
B({ \mathcal{E} \ |\ \omega _{\scriptsize L}}) = f^{T}({\mathcal{E} - \hbar\omega _{\scriptsize L}})\Big[ 1-f^{T}({\mathcal{E}}) \Big] - f^{T}(\mathcal{E})\Big[ 1- f^{T}(\mathcal{E}+\hbar\omega _{\scriptsize L}) \Big] 
\label{eq:thermal_factor_noneq_B_def}
\end{equation}

\begin{equation}
I(\hbar\omega_{E}|\hbar\omega_{L}) \propto \int\limits_{0}^{\infty} f(\mathcal{E}+\hbar\omega)[1-f(\mathcal{E})]d\mathcal{E}
\label{eq:thermal_factor_noneq_integral}
\end{equation}
